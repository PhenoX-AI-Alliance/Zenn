## 4. 繧ｳ繧｢繝ｭ繧ｸ繝・け縺ｮ螳溯｣・ｾ・
迺ｰ蠅・・螟牙喧縺ｫ蠑ｷ縺・ヱ繧､繝励Λ繧､繝ｳ繧堤ｶｭ謖√☆繧九◆繧√・繧ｳ繧｢繧ｳ繝ｼ繝峨・荳驛ｨ・・patch_pipeline.py`・峨〒縺吶・
```python
import os
import subprocess
import requests
from typing import Dict, Any, Optional
from github import Github

OLLAMA_HOST = os.getenv("OLLAMA_HOST", "http://localhost:11434")
OLLAMA_MODEL = os.getenv("OLLAMA_MODEL", "qwen2.5-coder:7b-instruct-q4_K_M")

class PatchPipelineEngine:
    def __init__(self):
        self.ollama_url = f"{OLLAMA_HOST}/api/generate"
        self.github_client = Github(os.getenv("GITHUB_TOKEN"))

    def analyze_and_generate_patch(self, error_context: Dict[str, Any]) -> Optional[str]:
        prompt = self._build_prompt(error_context)
        payload = {
            "model": OLLAMA_MODEL,
            "prompt": prompt,
            "stream": False,
            "options": {"temperature": 0.1, "num_ctx": 4096}
        }

        try:
            response = requests.post(self.ollama_url, json=payload, timeout=60)
            response.raise_for_status()
            raw_output = response.json().get("response", "")
            
            clean_diff = self._extract_diff(raw_output)
            
            if self._validate_diff_syntax(clean_diff, error_context["file_path"]):
                return clean_diff
            return None

        except requests.exceptions.RequestException as e:
            print(f"[ERROR] Ollama Backend Unreachable: {e}")
            return None

    def _extract_diff(self, llm_output: str) -> str:
        if "```diff" in llm_output:
            return llm_output.split("```diff")[1].split("```")[0].strip()
        return llm_output.strip()

    def run_sandbox_tests(self, patch_diff: str, workspace_dir: str) -> bool:
        patch_file = os.path.join(workspace_dir, "temp_fix.patch")
        with open(patch_file, "w") as f:
            f.write(patch_diff)

        try:
            chk = subprocess.run(["git", "apply", "--check", patch_file], cwd=workspace_dir, capture_output=True)
            if chk.returncode != 0: return False

            subprocess.run(["git", "apply", patch_file], cwd=workspace_dir, check=True)
            
            test_res = subprocess.run(["pytest", "tests/"], cwd=workspace_dir, capture_output=True)
            return test_res.returncode == 0
            
        except Exception as e:
            return False
        finally:
            subprocess.run(["git", "checkout", "."], cwd=workspace_dir, capture_output=True)
```
