## 3.2 コア実装：`core/ast_analyzer.py`（AST差分解析）
```python
import ast
import difflib
from typing import Dict, Tuple, List

class ASTDiffAnalyzer:
    """
    レガシーPythonモジュールの巨大ファイルを安全に解析し、
    脆弱性・依存性ドリフトに関連するノード差分のみを抽出するエンジン。
    """
    def __init__(self, max_loc_per_chunk: int = 1000):
        self.max_loc_per_chunk = max_loc_per_chunk

    def parse_safely(self, source_code: str) -> ast.Module:
        """OOMを防ぐため、極端に大きいコードの構文解析を安全に行う"""
        try:
            return ast.parse(source_code)
        except SyntaxError as e:
            raise ValueError(f"Legacy Syntax Error encountered: {e}")

    def extract_node_signatures(self, tree: ast.Module) -> List[Dict[str, any]]:
        """関数・クラス単位でノードを抽出し、メモリ肥大化を防止する"""
        nodes = []
        for node in ast.walk(tree):
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                nodes.append({
                    "name": node.name,
                    "lineno": node.lineno,
                    "end_lineno": getattr(node, "end_lineno", node.lineno),
                    "type": type(node).__name__
                })
        return nodes

    def generate_ast_diff_context(self, old_code: str, new_code: str) -> str:
        """人間が理解しやすい、かつLLMがコンテキスト迷子にならない最小限のDiff生成"""
        old_lines = old_code.splitlines()
        new_lines = new_code.splitlines()
        diff = difflib.unified_diff(
            old_lines, new_lines,
            fromfile='legacy_original.py',
            tofile='patched_candidate.py',
            lineterm=''
        )
        return "\n".join(list(diff)[:200]) # コンテキスト溢れを防ぐため最大200行に制限
```

## 3.3 コア実装：`core/sandbox_runner.py`（動的検証・三現主義の担保）
```python
import subprocess
import resource
import sys

class SandboxTestRunner:
    """
    LLMが生成した修正パッチを適用後、テストランナー（pytest）を安全に実行し、
    デッドロックやOOM（Exit 137）を検知・ハンドリングするサンドボックス。
    """
    def __init__(self, target_script: str, timeout_sec: int = 60):
        self.target_script = target_script
        self.timeout_sec = timeout_sec

    def _set_memory_limit(self):
        """子プロセスのメモリ暴走を防ぐため、Linux環境でリソース制限を設定（WSL/Linux前提）"""
        if sys.platform != "win32":
            # 例: 最大メモリを 2GB にハードリミット設定
            hard_limit = 2 * 1024 * 1024 * 1024
            resource.setrlimit(resource.RLIMIT_AS, (hard_limit, hard_limit))

    def run_verification(self) -> Tuple[bool, str]:
        """テストを実行し、生々しいエラーログ（終了コード含む）を返す"""
        try:
            result = subprocess.run(
                [sys.executable, "-m", "pytest", self.target_script, "--maxfail=1", "--disable-warnings"],
                capture_output=True,
                text=True,
                timeout=self.timeout_sec,
                preexec_fn=self._set_memory_limit if sys.platform != "win32" else None
            )
            
            if result.returncode == 0:
                return True, "Verification Passed Successfully."
            elif result.returncode == -9 or "Killed" in result.stderr:
                return False, "[CRITICAL] OOM Killer invoked: Sandbox detected memory threshold breach (Exit 137)."
            else:
                return False, f"Test Failed (Exit Code {result.returncode}):\n{result.stderr}"

        except subprocess.TimeoutExpired:
            return False, f"[CRITICAL] Deadlock detected: Pytest execution timed out after {self.timeout_sec} seconds."
        except Exception as e:
            return False, f"[ERROR] Sandbox execution failure: {str(e)}"
```
